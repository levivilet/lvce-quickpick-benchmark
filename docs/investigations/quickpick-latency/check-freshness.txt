import { readFile, writeFile, unlink } from 'node:fs/promises'
import { launch } from '../../packages/benchmark/src/launch.ts'
import { search } from '../../packages/benchmark/src/adapters.ts'
const editor=JSON.parse(await readFile('config/editors.lock.json','utf8'))[0]
const variant=process.argv[2]
const file='QuickpickFreshnessProbe.ts'
const app=await launch(editor,false,`results/freshness-${variant}.log`)
try {
 await search(app.page,'lvce','quickOpenModel.ts'); await app.page.keyboard.press('Escape')
 await writeFile(`.tmp/fixture/${file}`,'// temporary freshness probe\n',{flag:'wx'})
 await app.page.keyboard.press('Control+p')
 await app.page.locator('input[name="QuickPickInput"]').waitFor()
 await app.page.keyboard.insertText(file)
 let visible=false
 try {await app.page.locator('.QuickPickItemLabel').filter({hasText:file}).waitFor({timeout:5000});visible=true}catch{}
 await writeFile(`results/freshness-${variant}.json`,JSON.stringify({variant,file,visible,timeoutMs:5000},null,2))
 console.log({variant,visible})
} finally {await app.close();await unlink(`.tmp/fixture/${file}`).catch(()=>{})}
